// Definición de pines PWM permitidos
const int ledPins[] = {3, 5, 6, 9};

void setup() {
  Serial.begin(9600);
  for (int i = 0; i < 4; i++) {
    pinMode(ledPins[i], OUTPUT);
    analogWrite(ledPins[i], 0); // Inicializar apagados
  }
}

void loop() {
  if (Serial.available() > 0) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    cmd.toUpperCase(); // Para aceptar "led1" o "LED1"

    int separatorIndex = cmd.indexOf(':');
    
    // Validar que exista el separador ':' y que tenga el formato mínimo "LEDx:v"
    if (separatorIndex != -1) {
      String ledPart = cmd.substring(0, separatorIndex);      // Ejemplo: "LED1"
      String valPart = cmd.substring(separatorIndex + 1);     // Ejemplo: "255"
      
      int val = valPart.toInt();
      int pinIndex = -1;

      // Identificar qué LED se solicitó
      if (ledPart == "LED1") pinIndex = 0;
      else if (ledPart == "LED2") pinIndex = 1;
      else if (ledPart == "LED3") pinIndex = 2;
      else if (ledPart == "LED4") pinIndex = 3;

      // Validar pin y rango de valor PWM (0-255)
      if (pinIndex != -1 && val >= 0 && val <= 255) {
        analogWrite(ledPins[pinIndex], val);
        Serial.print("OK:");
        Serial.print(ledPart);
        Serial.print(":");
        Serial.println(val);
      } else {
        Serial.println("ERR:CMD");
      }
    } else {
      Serial.println("ERR:CMD");
    }
  }
}
